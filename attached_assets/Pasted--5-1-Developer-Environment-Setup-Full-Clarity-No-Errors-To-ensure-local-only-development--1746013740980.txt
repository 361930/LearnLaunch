require('dotenv').config();
const express = require('express');
const cors = require('cors');
const bodyParser = require('body-parser');
const app = express();
const PORT = process.env.PORT || 5000;

// Middleware
app.use(cors());
app.use(bodyParser.json());

// Auth endpoints
app.post('/auth/register', (req, res) => {
  console.log('Register:', req.body);
  res.status(200).send({ token: 'fake-jwt-token', role: req.body.role });
});

app.post('/auth/login', (req, res) => {
  console.log('Login:', req.body);
  res.status(200).send({ token: 'fake-jwt-token', role: req.body.role });
});

// Start server
app.listen(PORT, () => console.log(`API Server running at http://localhost:${PORT}`));

5.5 File-by-File Action Guide

File	Task
src/screens/LoginScreen.tsx	Build login form (email/password). On submit, call api.post('/auth/login'). Store JWT in AsyncStorage.
src/screens/RegisterScreen.tsx	Build register form. Call api.post('/auth/register'). Store JWT, then navigate based on role claim.
src/screens/DashboardStudent.tsx	Fetch enrolled classes from backend or local JSON. Display progress bars and recommendations.
src/screens/DashboardTeacher.tsx	Display created classes, enrollment stats. Add UI to create new classes (persist in memory or SQLite).
src/screens/ClassListScreen.tsx	Fetch all classes. Implement pagination with FlatList and lazy-loading (infinite scroll).
src/screens/ClassDetailScreen.tsx	Show class info: title, description, schedule calendar, join/enroll button.
src/screens/DiscussionScreen.tsx	Integrate Firebase Realtime DB: load and post messages, show timestamps, implement read receipts.
src/screens/ProfileScreen.tsx	Enable profile picture upload (use react-native-image-picker), bio editing, and logout button.
src/seed/demoUsers.ts	Define hidden demo accounts with roles:
